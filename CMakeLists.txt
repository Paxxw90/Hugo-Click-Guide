cmake_minimum_required(VERSION 3.25)
project(HugoClickGuide VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GUIDE_CORE_TESTS "Build standalone timing/parser tests without Geode" OFF)
if(GUIDE_CORE_TESTS)
    enable_testing()
    add_executable(guide_tests tests/core_tests.cpp)
    add_test(NAME guide_core COMMAND guide_tests)
    return()
endif()
if(NOT DEFINED ENV{GEODE_SDK})
    message(FATAL_ERROR "Set GEODE_SDK to the Geode SDK directory, or use the supplied GitHub Actions workflow.")
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
add_library(${PROJECT_NAME} SHARED src/main.cpp)
setup_geode_mod(${PROJECT_NAME})
